  <div id="adminPanel">
    <h3>⚙️ 管理者メニュー：新規教材・資料の追加</h3>
    <div class="form-row">
      <input type="text" id="inputYear" placeholder="開講年度 (例: 2026)" value="2026">
      <input type="text" id="inputSeason" placeholder="開講時期 (例: 通常授業 / 夏期講習)">
      <input type="text" id="inputCourse" placeholder="講座名 (例: 高3 ハイレベル物理)">
    </div>
    <div class="form-row">
      <input type="text" id="inputTitle" placeholder="タイトル (例: 第1回 力学・運動方程式)">
      <input type="text" id="inputPdf" placeholder="PDFのリンクURL">
      <input type="text" id="inputWord" placeholder="WordのリンクURL">
    </div>
    <div class="form-row">
      <input type="text" id="inputExcel" placeholder="ExcelのリンクURL">
      <input type="text" id="inputPpt" placeholder="PowerPointのリンクURL">
      <input type="text" id="inputVideo" placeholder="動画のリンクURL">
    </div>
    <button class="add-btn" onclick="addNewItem()">+ 講座・資料を登録する</button>
  </div>

  <!-- 絞り込みフィルター -->
  <div class="filter-box">
    <div class="filter-group">
      <label>開講年度で絞り込み</label>
      <select id="filterYear" onchange="render()"><option value="">すべて表示</option></select>
    </div>
    <div class="filter-group">
      <label>開講時期で絞り込み</label>
      <select id="filterSeason" onchange="render()"><option value="">すべて表示</option></select>
    </div>
    <div class="filter-group">
      <label>講座名で絞り込み</label>
      <select id="filterCourse" onchange="render()"><option value="">すべて表示</option></select>
    </div>
  </div>

  <!-- 資料リスト表示エリア -->
  <div id="listArea"></div>

  <script>
    // 初期サンプルデータ（必要に応じて編集・削除可能）
    const defaultItems = [
      { id: "1", year: "2026", season: "通常授業", course: "高3 ハイレベル物理", title: "第1回 力学・運動方程式", pdf: "#", word: "", excel: "", ppt: "", video: "#", published: true },
      { id: "2", year: "2026", season: "夏期講習", course: "高3 ハイレベル物理", title: "夏期特別演習：電磁気分野", pdf: "#", word: "#", excel: "", ppt: "#", video: "", published: true }
    ];

    let items = JSON.parse(localStorage.getItem("teacher_archive_items")) || defaultItems;
    let isAdmin = false;
    const ADMIN_PASS = "teacher2026"; // 共通パスワード（お好みで変更可能）

    function toggleAdminMode() {
      if (!isAdmin) {
        const pass = prompt("管理者パスワードを入力してください：");
        if (pass === ADMIN_PASS) {
          isAdmin = true;
          document.body.classList.add("admin-mode");
          document.getElementById("adminPanel").style.display = "block";
          document.getElementById("adminToggleBtn").textContent = "管理者ログアウト";
          document.getElementById("adminToggleBtn").classList.add("active");
          render();
        } else if (pass !== null) {
          alert("パスワードが違います。");
        }
      } else {
        isAdmin = false;
        document.body.classList.remove("admin-mode");
        document.getElementById("adminPanel").style.display = "none";
        document.getElementById("adminToggleBtn").textContent = "管理者ログイン";
        document.getElementById("adminToggleBtn").classList.remove("active");
        render();
      }
    }

    function updateDropdowns() {
      const years = [...new Set(items.map(i => i.year))].sort();
      const seasons = [...new Set(items.map(i => i.season))].sort();
      const courses = [...new Set(items.map(i => i.course))].sort();

      updateSelectOptions("filterYear", years);
      updateSelectOptions("filterSeason", seasons);
      updateSelectOptions("filterCourse", courses);
    }

    function updateSelectOptions(elementId, values) {
      const select = document.getElementById(elementId);
      const currentVal = select.value;
      let html = '<option value="">すべて表示</option>';
      values.forEach(v => {
        html += `<option value="${v}" ${v === currentVal ? 'selected' : ''}>${v}</option>`;
      });
      select.innerHTML = html;
    }

    function render() {
      updateDropdowns();
      const y = document.getElementById("filterYear").value;
      const s = document.getElementById("filterSeason").value;
      const c = document.getElementById("filterCourse").value;

      const filtered = items.filter(item => {
        // 管理者でない場合は非公開のものを表示しない
        if (!isAdmin && !item.published) return false;
        return (!y || item.year === y) &&
               (!s || item.season === s) &&
               (!c || item.course === c);
      });

      const area = document.getElementById("listArea");
      if (filtered.length === 0) {
        area.innerHTML = "<p style='text-align:center; color:#777; padding:20px;'>該当する資料はありません。</p>";
        return;
      }

      area.innerHTML = filtered.map(item => `
        <div class="card ${item.published ? 'published' : 'unpublished'}">
          <div>
            <div class="meta">${item.year}年度 / ${item.season} / ${item.course} ${!item.published ? '【非公開中】' : ''}</div>
            <div class="title">${item.title}</div>
          </div>
          <div style="display:flex; align-items:center;">
            <div class="links">
              <a href="${item.pdf || '#'}" class="pdf ${!item.pdf ? 'disabled' : ''}" target="_blank">PDF</a>
              <a href="${item.word || '#'}" class="word ${!item.word ? 'disabled' : ''}" target="_blank">Word</a>
              <a href="${item.excel || '#'}" class="excel ${!item.excel ? 'disabled' : ''}" target="_blank">Excel</a>
              <a href="${item.ppt || '#'}" class="ppt ${!item.ppt ? 'disabled' : ''}" target="_blank">PPT</a>
              <a href="${item.video || '#'}" class="video ${!item.video ? 'disabled' : ''}" target="_blank">動画</a>
            </div>
            <div class="admin-controls">
              <button class="toggle-pub" onclick="togglePublish('${item.id}')">${item.published ? '非公開にする' : '公開する'}</button>
              <button class="del-btn" onclick="deleteItem('${item.id}')">削除</button>
            </div>
          </div>
        </div>
      `).join("");
    }

    function addNewItem() {
      const year = document.getElementById("inputYear").value.trim();
      const season = document.getElementById("inputSeason").value.trim();
      const course = document.getElementById("inputCourse").value.trim();
      const title = document.getElementById("inputTitle").value.trim();

      if (!year || !season || !course || !title) {
        alert("「年度」「時期」「講座名」「タイトル」は必須入力です。");
        return;
      }

      const newItem = {
        id: Date.now().toString(),
        year: year,
        season: season,
        course: course,
        title: title,
        pdf: document.getElementById("inputPdf").value.trim(),
        word: document.getElementById("inputWord").value.trim(),
        excel: document.getElementById("inputExcel").value.trim(),
        ppt: document.getElementById("inputPpt").value.trim(),
        video: document.getElementById("inputVideo").value.trim(),
        published: true
      };

      items.push(newItem);
      saveAndRefresh();
      
      // 入力欄クリア
      document.getElementById("inputTitle").value = "";
      document.getElementById("inputPdf").value = "";
      document.getElementById("inputWord").value = "";
      document.getElementById("inputExcel").value = "";
      document.getElementById("inputPpt").value = "";
      document.getElementById("inputVideo").value = "";
      alert("新しい資料を追加しました！");
    }

    function togglePublish(id) {
      const target = items.find(i => i.id === id);
      if (target) {
        target.published = !target.published;
        saveAndRefresh();
      }
    }

    function deleteItem(id) {
      if (confirm("この資料データを削除してもよろしいですか？")) {
        items = items.filter(i => i.id !== id);
        saveAndRefresh();
      }
    }

    function saveAndRefresh() {
      localStorage.setItem("teacher_archive_items", JSON.stringify(items));
      render();
    }

    // 初期読み込み
    render();
  </script>
